\begin{frame}{Pregunta de investigación}
  \vspace{.45cm}
  \begin{block}{Pregunta general}
    \centering\large
    ¿Cuáles son los patrones de distribución espacial y temporal de las ocurrencias del extracto SIPCOP Municipal para San Sebastián, del 1 de enero al 29 de agosto de 2025?
  \end{block}
  \vspace{.3cm}
  \begin{itemize}
    \item ¿Qué proporción de los registros tiene ubicación, fecha y tipo utilizables?
    \item ¿Qué grupos detecta DBSCAN, cuáles superan lo esperado por azar y cuánto coincide HDBSCAN?
        \item ¿Cómo se distribuyen temporalmente las ocurrencias de los grupos significativos frente a la exposición? (fecha de alerta)
  \end{itemize}
\end{frame}

\begin{frame}{Objetivos e hipótesis}
  \begin{columns}[T,onlytextwidth]
    \column{.52\textwidth}
      \begin{block}{Objetivo general}
        Analizar la distribución espacial y temporal considerando la calidad y las limitaciones de los registros disponibles.
      \end{block}
      \textbf{Objetivos específicos}
      \begin{enumerate}
        \item Evaluar calidad y cobertura.
        \item Detectar y contrastar concentraciones con DBSCAN; comparar con HDBSCAN.
        \item Analizar la distribución temporal de las concentraciones significativas.
      \end{enumerate}
    \column{.44\textwidth}
      \begin{block}{Hipótesis de trabajo}
        \textbf{H1:} al menos un grupo espacial concentra más ocurrencias que las esperadas bajo aleatoriedad espacial completa.

        \medskip
        \textbf{H2:} en los grupos significativos, la distribución temporal difiere de la del resto del distrito.
      \end{block}
      \footnotesize El objetivo de calidad de datos es descriptivo y no requiere hipótesis.
  \end{columns}
\end{frame}

\begin{frame}{Métodos de agrupamiento espacial}
  \begin{columns}[T,onlytextwidth]
    \column{.48\textwidth}
      \begin{block}{DBSCAN: método principal}
        \begin{itemize}
          \item Agrupa \textbf{puntos densos} dentro de un radio $\varepsilon$.
          \item \textbf{Parámetros:} $\varepsilon$ y MinPts.
          \item Permite grupos de formas diversas y puntos aislados como \textbf{ruido}.
        \end{itemize}
      \end{block}
    \column{.48\textwidth}
      \begin{block}{HDBSCAN: comparación}
        \begin{itemize}
          \item Explora \textbf{jerarquías de densidad}.
          \item No requiere un \textbf{único $\varepsilon$ fijo}.
          \item Se compararán asignaciones, ruido y estabilidad.
        \end{itemize}
      \end{block}
  \end{columns}
  \vspace{.2cm}
  \footnotesize La densidad kernel se empleará solo como visualización complementaria, no como prueba de significancia. (Ester et al., 1996; Campello et al., 2013; McInnes et al., 2017.)
\end{frame}

  \end{block}
  \begin{itemize}
    \item Con coordenadas a 0,001° (${\sim}110$ m), \textbf{$\varepsilon=50$ y $100$ m dan la misma partición}; por eso la rejilla parte de \textbf{150 m}. Los puntos coincidentes cuentan por separado para MinPts.
    \item Se registrarán \textbf{número de grupos, porcentaje de ruido y silueta} (con cautela por los puntos coincidentes).
    \item La configuración se elegirá por \textbf{estabilidad} (ARI promedio frente a configuraciones vecinas) entre soluciones admisibles: al menos 2 grupos, menos de 60\,\% de ruido y ningún grupo con más de 25\,\% de los registros ni 1 km de diámetro.
    \item Se reportará \textbf{sensibilidad} a las demás configuraciones y \textbf{comparación con HDBSCAN}.
  \end{itemize}
  \vspace{.15cm}
  \footnotesize La regla se fijará antes de interpretar los resultados.
\end{frame}

% 16
\begin{frame}{Contraste espacial: ¿más concentración que por azar?}
  \begin{columns}[T,onlytextwidth]
    \column{.5\textwidth}
      \begin{block}{Modelo nulo}
        \textbf{Aleatoriedad espacial completa (CSR)}: $n$ puntos uniformes e independientes dentro del polígono distrital.
        \begin{itemize}
          \item \textbf{Simulación Monte Carlo:} $B=999$ simulaciones.
          \item Se repite el agrupamiento con los parámetros elegidos, redondeando los puntos simulados a la rejilla de 0,001°.
          \item \textbf{Segundo nulo:} área urbanizada (manzanas o vías). La CSR es solo la referencia mínima.
        \end{itemize}
      \end{block}
    \column{.46\textwidth}
      \begin{block}{Decisión por grupo $j$}
        Se compara el tamaño observado $m_j$ con el mayor grupo simulado $M_b$:
        \[
        p_j=\frac{1+\#\{b:M_b\geq m_j\}}{1+B}.
        \]
        Los grupos que superen el contraste se denominarán \textbf{\textit{hotspots}}.
      \end{block}
  \end{columns}
\end{frame}

% 17
\begin{frame}{Análisis temporal}
  \begin{block}{Qué se comparará}
    Distribución de cada hotspot por \textbf{turno, día de la semana y mes} frente al resto del distrito (\textbf{chi-cuadrado de homogeneidad} o permutación).
  \end{block}
  \begin{columns}[T,onlytextwidth]
    \column{.48\textwidth}
      \textbf{Análisis secundario: exposición calendario}
      \begin{itemize}
        \item Turno: horas de operación de cada turno.
        \item Día: cantidad de cada día en el periodo.
        \item Mes: número de días de cada mes. Se rechazará casi siempre: la tasa mensual va de 2,0 a 7,5 registros por día.
      \end{itemize}
    \column{.48\textwidth}
      \textbf{Pruebas y reporte}
      \begin{itemize}
        \item \textbf{Homogeneidad o permutación} como referencia principal; bondad de ajuste como secundaria.
        \item \textbf{Ajuste de Holm} por comparaciones múltiples.
        \item \textbf{V de Cramér} como tamaño del efecto.
      \end{itemize}
  \end{columns}
  \vspace{.15cm}
  \footnotesize Sensibilidad: solo registros ciudadanos (232), sin exposición a peligro o abandono, sin casi duplicados y con desplazamiento de $\pm55$ m. Los contrastes no prueban causalidad.
\end{frame}
